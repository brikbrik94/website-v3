# Info & Debug Page Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Create a new Info page with a sidebar for navigating between different info/debug modules. The first module is "NAH Status", showing operational times in a table.

**Architecture:** A centralized `InfoPage.ts` manages the layout and dynamic module loading. The router handles subpaths like `/info/nah`.

**Tech Stack:** TypeScript, CI Styles (Typ 1 & Typ 2).

// src/pages/InfoPage.ts
import { initTopbar } from '../components/Topbar';
import { navigate } from '../main';

export const initInfoPage = async (container: HTMLElement, subpath: string = 'nah') => {
  container.innerHTML = `
    <div id="topbar-mount"></div>
    <div class="layout">
      <aside class="sidebar">
        <div class="sidebar-inner">

          <a href="/info/nah" class="sidebar-nav-item ${subpath === 'nah' ? 'active' : ''}" data-module="nah">

          </a>
          <a href="/info/debug" class="sidebar-nav-item ${subpath === 'debug' ? 'active' : ''}" data-module="debug">

          <button class="sidebar-footer-copyright">©</button>
        </div>
      </aside>
      <main class="page-content" id="info-content-mount">
      </main>
    </div>
  `;

  // Initialize Topbar (minimal version for landing/info)
  const topbarMount = document.getElementById('topbar-mount')!;
  initTopbar(topbarMount, [], () => {}); // No basemaps needed for info

  // Module loading logic
  const contentMount = document.getElementById('info-content-mount')!;
  if (subpath === 'nah') {
    renderNahStatusModule(contentMount);
  } else {
    contentMount.innerHTML = '<div class="content-body"><div class="panel"><div class="panel-body">Coming soon...</div></div></div>';
  }
};
